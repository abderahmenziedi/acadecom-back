 * errorHandler.js — central Express error middleware.
 *
 * Translates anything thrown by a route into the canonical error envelope:
 *
 *   {
 *     success: false,
 *     status: "fail" | "error",
 *     message: string,
 *     error?: { code?: string, details?: any, stack?: string }
 *   }
 *
 * Handles:
 *  - `ApiError`            → faithful pass-through.
 *  - `ZodError`            → 422 with field-level details.
 *  - Prisma known errors   → mapped to 4xx where possible (P2002, P2003, P2025).
 *  - Everything else       → 500, generic message in production.
 *
 * Must be the last middleware in the stack.
 */

/**
 * Map a Prisma known error to an `ApiError`.
 * Returns null if the code is unrecognised — caller falls back to 500.
 * @param {any} err
 */
function mapPrismaError(err) {
  if (err?.name !== PRISMA_KNOWN_REQUEST_ERROR) return null;
  const target = err.meta?.target;
  switch (err.code) {
    case 'P2002':
      return ApiError.conflict('Cette valeur est déjà utilisée', {
        code: 'UNIQUE_CONSTRAINT',
        details: target ? { fields: Array.isArray(target) ? target : [target] } : null,
      });
    case 'P2003':
      return ApiError.badRequest('Référence externe invalide', {
        code: 'FOREIGN_KEY_VIOLATION',
        details: target ? { field: target } : null,
      });
    case 'P2025':
      return ApiError.notFound('Ressource introuvable', { code: 'NOT_FOUND' });
    default:
      return null;
  }
}
 
/**
 * Format a `ZodError` into a flat `[{ field, message }]` array.
 * @param {ZodError} err
 */
function zodIssues(err) {
  return err.issues.map((issue) => ({
    field: issue.path.join('.') || 'root',
    message: issue.message,
  }));
}
