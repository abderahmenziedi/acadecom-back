/**
 * envValidator.js — boot-time check of required env vars.
 *
 * Refuses obvious placeholders for `JWT_SECRET` (e.g. the one committed in
 * the legacy `.env`). In production, also enforces a minimum entropy of
 * 32 characters; in development, warns below 16.
 *
 * Exits with code 1 on any failure — the process must not start with
 * insecure credentials.
 */
 
const WEAK_SECRETS = new Set([
  'ton_secret_pour_jwt',
  'secret',
  'changeme',
  'jwt_secret',
  'jwt-secret',
  'your-jwt-secret',
]);
 
const REQUIRED = {
  DATABASE_URL: 'URL MySQL Prisma (mysql://user:pass@host:port/db)',
  JWT_SECRET: 'Secret HMAC pour signer les JWT (≥ 32 caractères en prod)',
  CLIENT_URL: 'Origine du frontend autorisé par CORS',
  NODE_ENV: 'development | production | test',
};
 
const OPTIONAL_DEFAULTS = {
  PORT: '3000',
  LOG_LEVEL: 'info',
  ACCESS_TOKEN_TTL: '4h',
  REFRESH_TOKEN_TTL: '7d',
  RATE_LIMIT_WINDOW_MS: '900000',
  RATE_LIMIT_MAX: '5',
};
 
/**
 * Validate process.env. Throws (via `process.exit(1)`) if any required var is
 * missing or insecure.
 *
 * @returns {void}
 */
function validateEnv() {
  const errors = [];
 
  for (const [key, description] of Object.entries(REQUIRED)) {
    const value = process.env[key];
    if (!value || String(value).trim().length === 0) {
      errors.push(`  ✗ ${key} manquant — ${description}`);
    }
  }
 
  const isProd = process.env.NODE_ENV === 'production';
  const secret = process.env.JWT_SECRET;
 
  if (secret) {
    if (WEAK_SECRETS.has(secret.toLowerCase())) {
      errors.push(
        `  ✗ JWT_SECRET utilise une valeur d'exemple (${secret}). Générez une vraie clé via \`node -e "console.log(require('crypto').randomBytes(48).toString('base64'))"\`.`,
      );
    } else if (isProd && secret.length < 32) {
      errors.push(`  ✗ JWT_SECRET trop court en production (${secret.length} caractères, requis ≥ 32).`);
    } else if (!isProd && secret.length < 16) {
      // Just a warning in dev — won't block startup.
      console.warn(
        `  ⚠ JWT_SECRET court (${secret.length} caractères). Acceptable en dev, mais générez-en un plus long avant déploiement.`,
      );
    }
  }
 
  for (const [key, fallback] of Object.entries(OPTIONAL_DEFAULTS)) {
    if (!process.env[key]) process.env[key] = fallback;
  }
 
  if (errors.length > 0) {
    /* eslint-disable no-console */
    console.error('\n🚨 Variables d’environnement invalides :');
    console.error(errors.join('\n'));
    console.error('\n📝 Vérifiez votre fichier .env (cf. .env.example)\n');
    /* eslint-enable no-console */
    process.exit(1);
  }
}
 
module.exports = { validateEnv, REQUIRED, OPTIONAL_DEFAULTS };
